VOZ DA GABI // ===================================== function selecionarVoz() { if (!("speechSynthesis" in window)) { return null; } const vozes = window.speechSynthesis.getVoices(); const vozesPortugues = vozes.filter(voz => voz.lang.toLowerCase() === "pt-br" ); // Prioriza vozes brasileiras com nomes conhecidos. const preferidas = [ "francisca", "luciana", "maria", "google português" ]; for (const nome of preferidas) { const encontrada = vozesPortugues.find(voz => voz.name.toLowerCase().includes(nome) ); if (encontrada) { return encontrada; } } return vozesPortugues[0] || null; } function falarTexto(texto) { if (!vozAtiva || !("speechSynthesis" in window)) { return; } window.speechSynthesis.cancel(); const textoLimpo = texto .replace(/\p{Extended_Pictographic}/gu, "") .replace(/\s+/g, " ") .trim(); if (!textoLimpo) return; const fala = new SpeechSynthesisUtterance(textoLimpo); fala.lang = "pt-BR"; fala.rate = 0.93; fala.pitch = 0.95; fala.volume = 1; const voz = selecionarVoz(); if (voz) { fala.voice = voz; } fala.onstart = () => { palcoPersonagem.classList.add("pensando"); atualizarStatus("Gabi está falando..."); }; fala.onend = () => { palcoPersonagem.classList.remove("pensando"); atualizarStatus("Gabi está aqui com você"); }; fala.onerror = () => { palcoPersonagem.classList.remove("pensando"); atualizarStatus("Gabi está aqui com você"); }; window.speechSynthesis.speak(fala); } if ("speechSynthesis" in window) { window.speechSynthesis.onvoiceschanged = () => { selecionarVoz(); }; } btnVoz.addEventListener("click", () => { vozAtiva = !vozAtiva; btnVoz.textContent = vozAtiva ? "🔊" : "🔇"; btnVoz.classList.toggle("mudo", !vozAtiva); btnVoz.setAttribute( "aria-label", vozAtiva ? "Desativar voz" : "Ativar voz" ); if (!vozAtiva && "speechSynthesis" in window) { window.speechSynthesis.cancel(); palcoPersonagem.classList.remove("pensando"); atualizarStatus("Gabi está aqui com você"); } });
